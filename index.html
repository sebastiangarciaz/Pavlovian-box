const cfg = {

  trialsPerPhase: 10,

  iti: 2000,

  usIntensity: 0.8,

  csOnMs: 800,

  usOnMs: 600,

  trialMs: 1600,

};


// ═══════════════════════════════════════════════════════

//  STATE

// ═══════════════════════════════════════════════════════

let protocol = 'cc';

let session = null;


const rat = {

  // Rescorla-Wagner associative strengths

  vCS:  0,   // CS+ → US association

  vCS2: 0,   // CS2 association (2nd-order)

  vINH: 0,   // CS- inhibitory strength


  totalTrials: 0,

  phase: 'idle',

  history:   [],   // { trial, v, vInh, v2, phase }

  crHistory: [],   // { trial, cr }


  anim: {

    salivation:  0,

    toneRing:    0,

    lightFlash:  0,

    foodVisible: false,

    inhibFlash:  0,

    cs2Ring:     0,

    mouth:       0,

    earPos:      0,

  }

};


// Rescorla-Wagner parameters

const RW = {

  alpha:  0.3,   // CS salience

  beta:   0.5,   // US salience

  lambda: 1.0,   // US asymptote

};


// ═══════════════════════════════════════════════════════

//  CANVAS SETUP

// ═══════════════════════════════════════════════════════

const THEORIES = {

  cc:  '<strong>Classical Conditioning:</strong> Repeated CS (tone) → US (food) pairings. The CR (licking/freezing) emerges as a prediction of the US. Rescorla-Wagner: ΔV = αβ(λ − V).',

  ext: '<strong>Respondent Extinction:</strong> CS presented without the US. Associative strength decreases (λ = 0). This is active inhibition, not forgetting — spontaneous recovery may occur.',

  s2:  '<strong>2nd-Order Conditioning:</strong> CS2 (light) is paired with CS1 (tone) that is already conditioned. CS2 acquires indirect associative strength without direct US contact. V(CS2) grows up to the limit of V(CS1).',

  inh: '<strong>Conditioned Inhibition:</strong> CS−/CS+ compound trials with no US. CS− develops a negative associative value (V < 0) that suppresses the CR when combined with CS+.',

};


function setProtocol(p) {

  protocol = p;

  document.querySelectorAll('.protocol-btn').forEach(b => b.classList.remove('active'));

  document.getElementById(`proto-${p}`).classList.add('active');

  document.getElementById('theory-box').innerHTML = THEORIES[p];

  setStatus(`Protocol: ${p.toUpperCase()} — Ready`);

}


// ═══════════════════════════════════════════════════════

//  SESSION RUNNER

// ═══════════════════════════════════════════════════════

async function runSession() {

  if (session) return;

  rat.phase = protocol;

  setStatus('Session in progress…');


  const n = cfg.trialsPerPhase;

  for (let t = 1; t <= n; t++) {

    if (!session && t > 1) break;

    if (t === 1) session = true;

    rat.totalTrials++;

    await runTrial(t, n);

    await sleep(cfg.iti);

  }


  session = null;

  setStatus(`Phase complete · ${n} trials`);

}


async function runTrial(t, total) {

  const a = rat.anim;

  let crStrength = 0;

  let logMsg = '';

  let logCls = '';


  updateStatUI(`Trial ${t}/${total}`, 'trial');

  document.getElementById('s-phase').textContent = phaseName(protocol);


  if (protocol === 'cc') {

    // CS onset

    a.toneRing = 1; a.earPos = 1;

    await sleep(cfg.csOnMs);

    // US onset

    a.foodVisible = true; a.mouth = 1;

    // Rescorla-Wagner update

    const deltaV = RW.alpha * RW.beta * (RW.lambda - rat.vCS);

    rat.vCS = Math.min(1, rat.vCS + deltaV);

    crStrength = Math.max(0, rat.vCS);

    a.salivation = crStrength;

    await sleep(cfg.usOnMs);

    a.foodVisible = false; a.toneRing = 0; a.earPos = 0;

    logMsg = `Trial ${t} | CS→US | V=${rat.vCS.toFixed(3)} | CR=${(crStrength*100).toFixed(0)}%`;

    logCls = 'log-cr';


  } else if (protocol === 'ext') {

    // CS alone — no US

    a.toneRing = 1; a.earPos = 0.5;

    crStrength = Math.max(0, rat.vCS);

    a.salivation = crStrength * 0.8;

    await sleep(cfg.csOnMs + cfg.usOnMs);

    // Extinction: λ = 0

    const deltaV = RW.alpha * RW.beta * (0 - rat.vCS);

    rat.vCS = Math.max(0, rat.vCS + deltaV);

    a.toneRing = 0; a.salivation = Math.max(0, rat.vCS);

    logMsg = `Extinction ${t} | CS alone | V=${rat.vCS.toFixed(3)} | CR↓`;

    logCls = 'log-ext';


  } else if (protocol === 's2') {

    // CS2 → CS1 (already conditioned)

    a.lightFlash = 1; a.cs2Ring = 1;

    await sleep(500);

    a.toneRing = 1;

    // CS1's strength acts as the effective λ for CS2

    const effectiveLambda = rat.vCS;

    const d2 = RW.alpha * 0.25 * (effectiveLambda - rat.vCS2);

    rat.vCS2 = Math.min(rat.vCS, Math.max(0, rat.vCS2 + d2));

    crStrength = rat.vCS2;

    a.salivation = crStrength;

    await sleep(cfg.usOnMs + 300);

    a.lightFlash = 0; a.cs2Ring = 0; a.toneRing = 0;

    logMsg = `2nd-Order ${t} | CS2→CS1 | V2=${rat.vCS2.toFixed(3)} | CR=${(crStrength*100).toFixed(0)}%`;

    logCls = 'log-s2';


  } else if (protocol === 'inh') {

    // CS+ / CS− compound — no US

    a.toneRing = 1;   // CS+

    a.inhibFlash = 1; // CS−

    await sleep(cfg.csOnMs);

    // Compound V = vCS + vINH; drive vINH negative

    const vSum = rat.vCS + rat.vINH;

    const dInh = RW.alpha * RW.beta * (0 - vSum);

    rat.vINH = Math.min(0, rat.vINH + dInh);

    crStrength = Math.max(0, rat.vCS + rat.vINH);

    a.salivation = crStrength;

    await sleep(cfg.usOnMs);

    a.toneRing = 0; a.inhibFlash = 0;

    logMsg = `Inhibition ${t} | CS+/CS−→no US | Vinh=${rat.vINH.toFixed(3)} | CR=${(crStrength*100).toFixed(0)}%`;

    logCls = 'log-inh';

  }


  // Record history

  rat.history.push({ trial: rat.totalTrials, v: rat.vCS, vInh: rat.vINH, v2: rat.vCS2, phase: protocol });

  rat.crHistory.push({ trial: rat.totalTrials, cr: crStrength * 100 });


  // Update stats

  document.getElementById('s-cr').textContent    = (crStrength * 100).toFixed(1);

  document.getElementById('s-assoc').textContent = rat.vCS.toFixed(3);

  document.getElementById('s-inh').textContent   = Math.abs(rat.vINH).toFixed(3);

  document.getElementById('s-total').textContent = rat.totalTrials;

  logEvent(logMsg, logCls);

  drawCharts();


  // Fade CR after trial

  setTimeout(() => { rat.anim.salivation *= 0.4; rat.anim.mouth = 0; }, 800);

}

function drawCRMeter(W, H, a) {

  const mx = 18, my = H * 0.1, mh = H * 0.55, mw = 12;

  roundRect2(ctx, mx, my, mw, mh, 4); ctx.fill();

  ctx.strokeStyle = '#b8a080'; ctx.lineWidth = 1; ctx.stroke();

  const fillH = mh * a.salivation;

  if (fillH > 0) {

    g.addColorStop(0, '#a01818'); g.addColorStop(1, '#d43030');

    ctx.fillStyle = g;

    roundRect2(ctx, mx, my + mh - fillH, mw, fillH, 4); ctx.fill();

  ctx.rotate(-Math.PI/2);

  ctx.fillStyle = '#5c3d1e';

  ctx.font = '8px Courier Prime, monospace';

  ctx.textAlign = 'center';

  ctx.font = '9px Courier Prime, monospace';

  ctx.textAlign = 'left';

  ctx.fillText(`${(a.salivation*100).toFixed(0)}%`, mx + mw + 4, my + mh - mh * a.salivation + 4);

  // ── EYES ──

  const eyeX = headX + 8, eyeY = headY - 8;

  ctx.fillStyle = '#f0e8d0';

  ctx.beginPath(); ctx.ellipse(eyeX, eyeY, 7, 5.5, 0.1, 0, Math.PI * 2); ctx.fill();

  const irisGrad = ctx.createRadialGradient(eyeX - 1, eyeY - 1, 0.5, eyeX, eyeY, 4.5);

  // ── INHIBITION INDICATOR ──

  if (isInhibited) {

    ctx.strokeStyle = `rgba(26,92,92,${0.35 + 0.35 * Math.sin(t * 4)})`;

    ctx.lineWidth = 2; ctx.setLineDash([4, 5]);

    ctx.beginPath(); ctx.ellipse(10, 0, bodyW + 18, bodyH + 18, 0, 0, Math.PI * 2); ctx.stroke();

    ctx.setLineDash([]);

  }


  ctx.restore();


  // ── STATE LABEL ──

  const stateLabel = isInhibited   ? 'Inhibited — no CR (CS−)' :

                     isEating      ? 'Consummatory response (US)' :

                     isAlert && crLevel > 0.4 ? 'CR: Anticipatory salivation' :

                     isAlert       ? 'Orienting response (CS)' :

                                     'Baseline state';

  ctx.fillStyle = '#8a6030';

  ctx.font = 'italic 9px Playfair Display, serif';

  ctx.textAlign = 'center';

  ctx.fillText(`Experimental dog — ${stateLabel}`, cx + 14, cy + 64);

}

function drawAcqChart() {

  const cw = aqc.width, ch = aqc.height;

  aqCtx.clearRect(0, 0, cw, ch);

  aqCtx.fillStyle = '#f5f0e8'; aqCtx.fillRect(0, 0, cw, ch);


  const pad = { l:42, r:12, t:10, b:24 };

  const gw = cw - pad.l - pad.r, gh = ch - pad.t - pad.b;


  aqCtx.strokeStyle = 'rgba(184,160,128,0.35)'; aqCtx.lineWidth = 0.5;

  for (let i = 0; i <= 4; i++) {

    const y = pad.t + gh * (i/4);

    aqCtx.beginPath(); aqCtx.moveTo(pad.l, y); aqCtx.lineTo(pad.l+gw, y); aqCtx.stroke();

    aqCtx.fillStyle = '#8a6a4a'; aqCtx.font = '8px Courier Prime, monospace'; aqCtx.textAlign = 'right';

    aqCtx.fillText((1-i/4).toFixed(1), pad.l-4, y+3);

  }


  const h = rat.history;

  if (h.length < 2) return;

  const maxT = h[h.length-1].trial;

  const tx = t => pad.l + ((t-1)/(Math.max(maxT,2)-1)) * gw;

  const vy = v => pad.t + gh * (1 - v);


  [

    { key:'v',    color:'#1a3a8a', label:'V(CS+)' },

    { key:'v2',   color:'#6a3a9a', label:'V(CS2)' },

    { key:'vInh', color:'#1a5c5c', label:'|V(CS−)|', abs:true },

  ].forEach(({ key, color, abs }) => {

    const pts = h.filter(d => {

      if (key === 'v2')   return d.phase === 's2';

      if (key === 'vInh') return d.phase === 'inh';

      return true;

    });

    if (pts.length < 2) return;

    aqCtx.beginPath(); aqCtx.strokeStyle = color; aqCtx.lineWidth = 1.8;

    pts.forEach((d, i) => {

      const val = abs ? Math.abs(d[key]) : d[key];

      i === 0 ? aqCtx.moveTo(tx(d.trial), vy(val)) : aqCtx.lineTo(tx(d.trial), vy(val));

    });

    aqCtx.stroke();

    pts.forEach(d => {

      const val = abs ? Math.abs(d[key]) : d[key];

      aqCtx.fillStyle = color;

      aqCtx.beginPath(); aqCtx.arc(tx(d.trial), vy(val), 2.5, 0, Math.PI*2); aqCtx.fill();

    });

  });


  aqCtx.strokeStyle = 'rgba(92,61,30,0.4)'; aqCtx.lineWidth = 1;

  aqCtx.beginPath(); aqCtx.moveTo(pad.l, pad.t+gh); aqCtx.lineTo(pad.l+gw, pad.t+gh); aqCtx.stroke();

  aqCtx.fillStyle = '#5c3d1e'; aqCtx.font = '8px Courier Prime, monospace'; aqCtx.textAlign = 'center';

  aqCtx.fillText('Trials →', pad.l+gw/2, ch-4);

}


function drawCRChart() {

  const cw = crc.width, ch = crc.height;

  crCtx.clearRect(0, 0, cw, ch);

  crCtx.fillStyle = '#f5f0e8'; crCtx.fillRect(0, 0, cw, ch);


  const pad = { l:42, r:12, t:10, b:24 };

  const gw = cw - pad.l - pad.r, gh = ch - pad.t - pad.b;


  crCtx.strokeStyle = 'rgba(184,160,128,0.35)'; crCtx.lineWidth = 0.5;

  for (let i = 0; i <= 4; i++) {

    const y = pad.t + gh * (i/4);

    crCtx.beginPath(); crCtx.moveTo(pad.l, y); crCtx.lineTo(pad.l+gw, y); crCtx.stroke();

    crCtx.fillStyle = '#8a6a4a'; crCtx.font = '8px Courier Prime, monospace'; crCtx.textAlign = 'right';

    crCtx.fillText(`${100-i*25}%`, pad.l-4, y+3);

  }


  const d = rat.crHistory;

  if (d.length < 2) return;

  const maxT = d[d.length-1].trial;

  const tx = t => pad.l + ((t-1)/Math.max(maxT-1,1)) * gw;

  const cy2 = v => pad.t + gh * (1 - v/100);


  crCtx.beginPath();

  crCtx.moveTo(tx(d[0].trial), pad.t+gh);

  d.forEach(pt => crCtx.lineTo(tx(pt.trial), cy2(pt.cr)));

  crCtx.lineTo(tx(d[d.length-1].trial), pad.t+gh);

  crCtx.closePath();

  const fillG = crCtx.createLinearGradient(0, pad.t, 0, pad.t+gh);

  fillG.addColorStop(0, 'rgba(160,24,24,0.25)'); fillG.addColorStop(1, 'rgba(160,24,24,0.02)');

  crCtx.fillStyle = fillG; crCtx.fill();


  crCtx.beginPath(); crCtx.strokeStyle = '#a01818'; crCtx.lineWidth = 2;

  d.forEach((pt, i) => {

    i === 0 ? crCtx.moveTo(tx(pt.trial), cy2(pt.cr)) : crCtx.lineTo(tx(pt.trial), cy2(pt.cr));

  });

  crCtx.stroke();


  crCtx.strokeStyle = 'rgba(92,61,30,0.4)'; crCtx.lineWidth = 1;

  crCtx.beginPath(); crCtx.moveTo(pad.l, pad.t+gh); crCtx.lineTo(pad.l+gw, pad.t+gh); crCtx.stroke();

  crCtx.fillStyle = '#5c3d1e'; crCtx.font = '8px Courier Prime, monospace'; crCtx.textAlign = 'center';

  crCtx.fillText('Trials → (CR%)', pad.l+gw/2, ch-4);

}

function roundRect2(ctx, x, y, w, h, r) {

  ctx.beginPath();

  ctx.moveTo(x+r, y); ctx.lineTo(x+w-r, y); ctx.quadraticCurveTo(x+w, y, x+w, y+r);

  ctx.lineTo(x+w, y+h-r); ctx.quadraticCurveTo(x+w, y+h, x+w-r, y+h);

  ctx.lineTo(x+r, y+h); ctx.quadraticCurveTo(x, y+h, x, y+h-r);

  ctx.lineTo(x, y+r); ctx.quadraticCurveTo(x, y, x+r, y);

  ctx.closePath();

}
